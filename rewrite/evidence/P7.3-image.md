The construct image is the cached `jackin.invalid/construct:p72` (id `sha256:0269cd32b45de3ed91e46a3ea725822322002df9ca212b99b4d802ba6fcdf1af`, built from the unchanged construct Dockerfile by `/private/tmp/claude-501/p72/build.sh`). Scripts and raw logs: `/private/tmp/claude-501/p73-live/` (`lib.sh`, `sb.sh`, `L*.sh`, `L*.out`).

## Helpers

`lib.sh` defines `B` (`docker buildx build --builder orbstack --platform linux/arm64 --provenance=false --sbom=false --load`), `role NAME` (a role image `FROM construct` with the body on stdin), `derived NAME SRC [args]` (the derived build with `--build-context role=docker-image://...`, the Codex build args of the `AgentSpec` row and a context of `Dockerfile`, `jackin-start`, `scan` and two empty hook files) and the hostile roles:

- H1: `FROM construct`, `USER 0`, plants `capsuid` (mode 4755) and `capbin` (`setcap cap_net_raw+ep`), replaces `getcap` with `exit 0` and `find` with a wrapper that drops `capsuid` from the output.
- H3: H1 plus `/bin/sh` replaced by `#!/bin/bash` + `: > /verdict` + `exit 0`, so every `RUN` that uses the role shell succeeds without running (and leaves an empty verdict file).

```sh
  B -f "$d/Dockerfile" --build-context construct=docker-image://$C -t $P/$1-role "$d" > "$W/$1.role.log" 2>&1; echo "role $1 exit=$?"
}
mkctx() {
  c=$W/ctx-$1; rm -rf "$c"; mkdir -p "$c/hooks"
  cp "$IMG/derived/Dockerfile" "$IMG/derived/scan" "$IMG/jackin-start" "$c"/
  printf '#!/bin/sh\necho setup\n' > "$c/hooks/setup"; printf '#!/bin/sh\necho preflight\n' > "$c/hooks/preflight"
}
derived() {
  n=$1; mkctx "$n"; [ -z "${2:-}" ] || cp "$2" "$c/Dockerfile"
  B -f "$c/Dockerfile" --build-context construct=docker-image://$C --build-context role=docker-image://$P/$n-role \
    --build-arg CODEX_VERSION=0.166.0 --build-arg CODEX_BIN_DIR=/usr/local/bin \
    --build-arg CODEX_SHA256_ARM64=$ARM --build-arg CODEX_SHA256_AMD64=$AMD \
    -t $P/$n "$c" > "$W/$n.log" 2>&1
  rc=$?; echo "derived $n exit=$rc"
}
sha() { shasum -a 256 "$1" | cut -d' ' -f1; }
final_check() {
  n=$1; c=$W/ctx-$n
  if [ "$rc" -ne 0 ]; then
    echo "OUTCOME $n refused:"; grep -hE 'differs|protected path|account|uid 0|device node|setuid|capabilit|forbidden' "$W/$n.log" | sed 's/^#[0-9]* [0-9.]* //' | sort -u | head -8
    return 0
  fi
  docker save -o "$W/$n.tar" "$P/$n" || { echo "OUTCOME $n save failed"; return 1; }
  python3 -I "$T/final.py" "$W/$n.tar" \
    "sha:/usr/local/bin/jackin-start=$(sha "$c/jackin-start")" \
    "sha:/usr/local/lib/jackin/hooks/setup=$(sha "$c/hooks/setup")" \
    "sha:/usr/local/lib/jackin/hooks/preflight=$(sha "$c/hooks/preflight")" dirs nodev
  r=$?; rm -f "$W/$n.tar"
  if [ $r -eq 0 ]; then echo "OUTCOME $n built, state clean"; else echo "OUTCOME $n BREACH"; fi
  return 0
}
clean() { docker image ls --format '{{.Repository}}:{{.Tag}}' | grep "^$P/" | while read -r t; do docker image rm "$t" >/dev/null; done; }
```

### Helper `final.py`

```python
import hashlib, io, json, sys, tarfile

DIRS = ["/usr", "/usr/local", "/usr/local/bin", "/usr/local/lib",
        "/usr/local/lib/jackin", "/usr/local/lib/jackin/hooks"]
tar_path, specs = sys.argv[1], sys.argv[2:]
wanted = {s.split(":", 1)[1].split("=")[0] for s in specs if s.startswith(("sha:", "show:"))}
outer = tarfile.open(tar_path)
layers = json.load(outer.extractfile("manifest.json"))[0]["Layers"]
fs = {}
for name in layers:
    layer = tarfile.open(fileobj=io.BytesIO(outer.extractfile(name).read()))
    for m in layer.getmembers():
        n = m.name[2:] if m.name.startswith("./") else m.name
        path = "/" + n.rstrip("/")
        head, tail = path.rsplit("/", 1)
        if tail == ".wh..wh..opq":
            for k in [k for k in fs if k.startswith(head + "/")]:
                del fs[k]
            continue
        if tail.startswith(".wh."):
            gone = head + "/" + tail[4:]
            for k in [k for k in fs if k == gone or k.startswith(gone + "/")]:
                del fs[k]
            continue
        body = layer.extractfile(m).read() if m.isfile() and path in wanted else None
        fs[path] = (m, body)
failed = False
def say(ok, text):
    global failed
    failed = failed or not ok
    print(("PASS " if ok else "FAIL ") + text)
for spec in specs:
    kind, _, rest = spec.partition(":")
    if kind == "sha":
        path, _, want = rest.partition("=")
        m, body = fs.get(path, (None, None))
        got = hashlib.sha256(body).hexdigest() if body is not None else "none"
        say(m is not None and m.isfile() and got == want and (m.uid, m.gid) == (0, 0) and not m.mode & 0o022,
            f"{path} {oct(m.mode) if m else 'missing'} sha {got[:12]} want {want[:12]}")
    elif kind == "dirs":
        for d in DIRS:
            m = fs.get(d, (None, None))[0]
            say(m is not None and m.isdir() and (m.uid, m.gid) == (0, 0) and not m.mode & 0o022,
                f"{d} {oct(m.mode) if m else 'missing'} {(m.uid, m.gid) if m else ''}")
    elif kind == "nodev":
        bad = sorted(p for p, (m, _) in fs.items() if m.isblk() or m.ischr())
        say(not bad, f"device nodes {bad}")
    elif kind == "show":
        m, body = fs.get(rest, (None, None))
        print(rest + ": " + ("missing" if m is None else ("symlink -> " + m.linkname if m.issym() else (body or b"").decode(errors="replace"))))
sys.exit(1 if failed else 0)
```
